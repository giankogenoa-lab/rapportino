function generaRiepilogo() { const note = document.getElementById("note") .value .trim(); const agentiInseriti = document.querySelectorAll(".blocco-agente"); if (agentiInseriti.length === 0) { alert("Inserire almeno un agente."); return; } if (note === "") { alert("Compilare il campo Note."); return; } let html = `

Data riferimento: ${document.getElementById("dataRiferimento").value}

Richiedente: ${document.getElementById("cognomeRichiedente").value} ${document.getElementById("nomeRichiedente").value}

Agenti

`; agentiInseriti.forEach(agente => { const cognome = agente.querySelector(".cognomeAgente").value; const nome = agente.querySelector(".nomeAgente").value; const cid = agente.querySelector(".cidAgente").value; const turno = agente.querySelector(".turno").value; if (turno === "") { alert( "Tutti gli agenti devono avere un turno." ); throw ""; } html += `

${cognome} ${nome}
CID: ${cid}
Turno: ${turno}

`; }); html += `

Note

${note}

`; document .getElementById("contenutoRiepilogo") .innerHTML = html; document .getElementById("paginaInserimento") .style.display = "none"; document .getElementById("paginaRiepilogo") .style.display = "block"; } document.addEventListener("DOMContentLoaded", () => { const btnModifica = document.getElementById("modifica"); if(btnModifica){ btnModifica.addEventListener("click", () => { document .getElementById("paginaRiepilogo") .style.display = "none"; document .getElementById("paginaInserimento") .style.display = "block"; }); } });
